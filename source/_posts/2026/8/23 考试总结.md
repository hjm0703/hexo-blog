**T2** 非常好些，直接就 AC 了。

然后 **T3** 的贪心先验证一下之后题目就变成了需要每一次修改一个点的权值，然后每一次还需要按照权值排序。

然后我就发现就变成了一个区间平移，还有一些区间乘法。然后都需要平移了，那么直接使用平衡树就可以了。然后就调不出来了。

但是实际上这里唯一需要平衡树的理由是需要插入元素，但是可以离线下来，这样提前预留位置就可以了。

**T4** 直接写暴力，然后也不知道为什么 $  0$ 了。

static constexpr double eps = 1e-6;
static constexpr int N = 205;
int n, k, r;

int cnt[N];
ld vx[N], vy[N];

struct Node {
    int x, y, c;
} a[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

void update() {
    memset(vx, 0, sizeof vx);
    memset(vy, 0, sizeof vy);
    memset(cnt, 0, sizeof cnt);
  
    for(int i=1; i<=n; i++) {
        vx[a[i].c] += a[i].x, vy[a[i].c] += a[i].y;
        cnt[a[i].c]++;
    }
  
    for(int i=1; i<=k; i++) if(cnt[i]) vx[i] /= cnt[i], vy[i] /= cnt[i];
}

ld dis(int x, int y, int c) {
    ld dx = fabs(x - vx[c]);
    ld dy = fabs(y - vy[c]);
    return dx * dx + dy * dy;
}

int ksm(int x, int y) {
    int ans = 1;
    while(y) {
        if(y&1) ans = 1ll * ans * x % mod;
        x = 1ll * x * x % mod, y >>= 1;
    }
    return ans;
} 

struct SegmentTree {
    #define lc (p << 1)
    #define rc (p << 1 | 1)

    int sum[N<<2], tag[N<<2];

    void build(int p, int l, int r, int* a) {
        tag[p] = 1;
        if(l == r) return sum[p] = a[l], void();
        int mid = (l + r) >> 1;
        build(lc, l, mid, a), build(rc, mid + 1, r, a);
        sum[p] = (sum[lc] + sum[rc]) % mod;
    }

    void push_down(int p) {
        if(tag[p] != 1) {
            sum[lc] = 1ll * sum[lc] * tag[p] % mod;
            sum[rc] = 1ll * sum[rc] * tag[p] % mod;
            tag[lc] = 1ll * tag[lc] * tag[p] % mod;
            tag[rc] = 1ll * tag[rc] * tag[p] % mod;
            tag[p] = 1;
        }
    }
  
    void change(int p, int l, int r, int x, int y, int k) {
        if(x > y) return;
        if(x <= l && r <= y) {
            sum[p] = 1ll * sum[p] * k % mod;
            tag[p] = 1ll * tag[p] * k % mod;
            return ;
        }

        int mid = (l + r) >> 1;
        push_down(p);
        if(x <= mid) change(lc, l, mid, x, y, k);
        if(y > mid) change(rc, mid + 1, r, x, y, k);
        sum[p] = (sum[lc] + sum[rc]) % mod;
    }

    void change2(int p, int l, int r, int x, int k) {
        if(l == r) {
            sum[p] = k % mod;
            return ;
        }

        int mid = (l + r) >> 1;
        push_down(p);
        if(x <= mid) change2(lc, l, mid, x, k);
        else change2(rc, mid + 1, r, x, k);
        sum[p] = (sum[lc] + sum[rc]) % mod;
    }
}T;

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 2e5 + 5;
int n, m;
int a[N], b[N];
int aa[N], bb[N];
int u[N], v[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

int ans = 0;
bool check(int* c) {
    for(int i=1; i<=n; i++) if(c[i] != b[i]) return false;
    return true;
}

ll decom(int* c) {
    ll res = 0;
    for(int i=1; i<=n; i++) res = res * 10 + c[i];
    return res;
}

unordered_map<ll, bool> vis;

void dfs(int x, int c[]) {
    if(x >= ans) return ;
    if(check(c)) return ans = min(x, ans), void();
    if(x == ans) return ;
    if(vis.count(decom(c))) return;
    vis[decom(c)] = true;
    for(int i=1; i<=m; i++) {
        swap(c[u[i]], c[v[i]]);
        dfs(x+1, c);
        swap(c[u[i]], c[v[i]]);
    }
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

    #ifdef LOCAL

    #endif

    freopen("network.out", "w", stdout);

    if(n > 9) {

        return 0;
    }
  
    for(int i=1; i<=n; i++) cin >> aa[i], a[aa[i]] = i;
    for(int i=1; i<=n; i++) cin >> bb[i], b[bb[i]] = i;
