#include <bits/stdc++.h>
using namespace std;

using ll = long long;
#define int long long

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 2e5 + 5;
int n, k, q, mx;
int h[N], rk[N];
int c[N][10], nxt[8][N][18];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

ll gcd(ll a, ll b) { return b == 0 ? a : gcd(b, a % b); }
ll lcm(ll a, ll b) { return a / gcd(a, b) * b; }

inline int jump(int x, ll k, int c) {
    if(k > mx) return nxt[c][x][17];
    for(int i=17; i>=0; i--) {
        if(k >= (1 << i)) {
            k -= (1 << i);
            x = nxt[c][x][i];
        }
    }
    return x;
}

static constexpr int S = 1 << 20;
char buf[S], *p1 = buf, *p2 = buf;
inline char getc() {
    return p1 == p2 && (p2 = (p1 = buf) + fread(buf, 1, S, stdin), p1 == p2) ? EOF : *p1++;
}

inline int read() {
    int x = 0; char ch = getc();
    while(ch != EOF && (ch < '0' || ch > '9')) ch = getc();
    while(ch >= '0' && ch <= '9') x = (x << 3) + (x << 1) + (ch ^ 48), ch = getc();
    return x;
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
  
    #ifdef LOCAL
    freopen("samples/input2.txt", "r", stdin);
    // freopen("samples/rotation2.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("rotation.in", "r", stdin);
    freopen("rotation.out", "w", stdout);

    n = read(), k = read(), q = read();
    for(int i=1; i<=n; i++) h[i] = read(), mx = max(mx, h[i]), rk[i] = h[i] % k;
    for(int i=1; i<=n; i++) {
        for(int j=0; j<k; j++) {
            c[i][j] = read();
            if(c[i][j] == 0) c[i][j] = i;
        }
        for(int j=0; j<k; j++)
            nxt[j][i][0] = c[i][(rk[i] + j) % k];
    }

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
  
    for(int t=0; t<k; t++) {
        for(int i=1; i<18; i++) {
            for(int j=1; j<=n; j++)
                nxt[t][j][i] = nxt[t][nxt[t][j][i-1]][i-1];
        }
    }

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
  
    for(int t=1; t<=q; t++) {
        int x = read(), y = read(), a = read(), b = read();
        int cx = (k - rk[x]) % k, cy = (k - rk[y]) % k; 
      
        if(x == y) {
            cout << 0 << '\n';
        } else {
            int l = 1, r = max(mx / a, mx / b) + 100, ans = -1;
          
            while(l <= r) {
                int mid = (l + r) >> 1;
                if(jump(x, mid * a, cx) == jump(y, mid * b, cy)) {
                    ans = mid, r = mid - 1;
                }else l = mid + 1;
            }

            if(ans != -1) {
                for(int i=max(0ll, ans-100); i<ans; i++) {
                    if(jump(x, i * a, cx) == jump(y, i * b, cy)) {
                        ans = i;
                        break;
                    }
                }
                cout << ans << '\n';
            } else cout << ans << '\n';
        }
    }

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using ull = unsigned long long;

#define int long long

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 1e6 + 5;
int n;
int x[N], y[N], d[N], ans[N];

bool vis_[N<<5];

const int dx[] = {-1, 1, 0, 0, -1, 1, -1, 1};
const int dy[] = {0, 0, -1, 1, -1, -1, 1, 1};
/*~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct myhash {
    static inline ull mix(ull x) {
        x = ((x + 1) << 18) & 0x3247853678;
        x = ((x + 1) >> 18) & 0x3875446756;
        return x ^ (x >> 1);
    }

    ull operator () (pair<int, int> x) const {
        // static ull time = chrono::system_clock::now().time_since_epoch().count();
        return x.first + (x.second + 1) * 1000000000ull;
    }
};

int cnt = 0;
unordered_map<pair<int, int>, int, myhash> mp, vis;
inline int Getid(int x, int y) {
    if(!mp.count({x, y})) return mp[{x, y}] = ++cnt;
    else return mp[{x, y}];
}

struct DSU {
    int sum = 0;
    int fa[N << 4];

    void build(int n) {
        for(int i=1; i<=n; i++) fa[i] = i;
    }

    int find(int x) {
        return fa[x] == x ? x : fa[x] = find(fa[x]);
    }

    void merge(int x, int y) {
        x = find(x), y = find(y);
        if(x == y) return ;
        fa[x] = y, sum--;
    }
}dsu;

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    #ifdef LOCAL
    freopen("samples/input.txt", "r", stdin);
    freopen("samples/lake1.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("lake.in", "r", stdin);
    freopen("lake.out", "w", stdout);

    cin >> n;
    for(int i=1; i<=n; i++) cin >> x[i] >> y[i];

    int cnt = 0;
    for(int i=1; i<=n; i++) {
        if(vis.count({x[i], y[i]})) continue;
        Getid(x[i], y[i]);
        for(int j=0; j<8; j++) {
            int nx = x[i] + dx[j], ny = y[i] + dy[j];
            Getid(nx, ny);
        }
        vis[{x[i], y[i]}] = i, d[i] = 1;
        cnt++;
    }

    dsu.build(::cnt), dsu.sum = ::cnt - cnt;
  
    for(auto item : mp) {
        int x = item.first.first, y = item.first.second;
        if(vis.count({x, y})) continue;
        for(int j=0; j<8; j++) {
            int nx = x + dx[j], ny = y + dy[j];
            if(!mp.count({nx, ny})) continue;
            if(vis.count({nx, ny})) continue;
            dsu.merge(Getid(x, y), Getid(nx, ny));
        }
    }
  
    for(int i=n; i>=1; i--) if(d[i]) {
        ans[i] = dsu.sum - 1;
        break;
    }
  
    for(int i=n; i>=1; i--) if(d[i]) {
        dsu.sum++;
        for(int j=0; j<8; j++) {
            int nx = x[i] + dx[j], ny = y[i] + dy[j];
            if(vis.count({nx, ny}) && vis[{nx, ny}] != 0 && vis[{nx, ny}] < i) {
                continue;
            }
            dsu.merge(Getid(x[i], y[i]), Getid(nx, ny));
        }
        ans[i-1] = dsu.sum - 1;
    }

    for(int i=1; i<=n; i++) if(!d[i]) ans[i] = ans[i-1];


    int cnt2 = 0;
    for(int i=1; i<=::cnt; i++) {
        if(!vis_[dsu.find(i)]) {
            cnt2++;
            vis_[dsu.find(i)] = 1;
        }
    }

    for(int i=1; i<=n; i++) ans[i] -= cnt2 - 1;

    for(int i=1; i<=n; i++) cout << ans[i] << '\n';

    cerr << 1.0 * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}
