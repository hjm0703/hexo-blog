## 题目概述

**省流:**  $ 90 + 45 + 40 + 45 = 220$

首先 **T1** 写起来思虑还是非常明显的，我用了一种更加暴力的做法，就是去除每一个点之后再 DP 一次。最后坠机的 $ 10$ 分，原因是我对于 $ root$ 存在一个特判，而他依赖于 DFS 所存贮的左右儿子。而我计算每一个位置的时候还再次 DFS 了一次，导致有极小概率会破坏掉 son 的存储。

然后 **T2** 第一眼就直接断定容斥加上大粪讨。 然后我一算基础容斥的时间复杂度，$\mathcal O(n^6)$ 这不坠机，然后就去打部分分了。然后后来为了打最后一个部分分，这里可以发现上面的时间复杂度感觉绰绰有余。然后就被容斥肘击了。~~目前感觉考场上写容斥都没有写出来过~~。但是实际上这道题更加优秀的做法是轮廓线 DP。 他不需要考虑各种 conner case。 赛后 30 分钟就写完了。实际上最开始看到那个数据范围，也大概想过，但是问题是他一共就放三个 `L` 再图上，这不是明显的大粪套吗。结果 $ 3$ 是帮助轮廓线证明时间复杂度的。关键是谁能想到 NOIP 模拟赛靠轮廓线啊。

**T3， T4** 爽爽吃部分分，然后可以注意到 T4 如果加一个 `ios` 可以多得 15 分。

        for(int j=i; j<=i+n-1; j++) {

            cnt[s[j] - 'a']++;
        }
        if(sum > d) return false;

    return true;

        return ;
    }

    for(char i='a'; i<='z'; i++) {
        s.push_back(i);
        dfs(x + 1, s);
        s.pop_back();
    }
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 1e6 + 5;
int n, m, k;

struct Node {
    int x, y, z;
} a[N];

struct Edge {
    int x, y, l;
} b[N];

pair<int, int> que[N];

int end_time[N], ans[N];

double dis(Node a, Node b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    double dz = a.z - b.z;
    return sqrt(dx * dx + dy * dy + dz * dz);
}

vector<pair<int, int>> v[N];
vector<int> g[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct DSU {
    int fa[N];

    void build(int n) {
        for(int i=1; i<=n; i++) fa[i] = i;
    }

    int find(int x) {
        return x == fa[x] ? x : fa[x] = find(fa[x]);
    }

    void merge(int x, int y) {
        x = find(x), y = find(y);
        if(x == y) return ;
        fa[x] = y;
    }
} dtr;

void solve() {
    cin >> n;
    for(int i=1; i<=n; i++) cin >> a[i].x >> a[i].y;
    for(int i=1; i<=n; i++) v[i].clear();
    cin >> m;
    memset(end_time, -1, sizeof end_time);
    for(int i=1; i<=m; i++) {
        int x, y, l; cin >> x >> y >> l;
        v[x].emplace_back(y, i);
        v[y].emplace_back(x, i);
        b[i] = Edge{x, y, l};
        if(dis(a[x], a[y]) > l) end_time[i] = 0;
    }

    cin >> k;
    for(int i=1; i<=k; i++) {
        g[i].clear();
        int x, h; cin >> x >> h;
        a[x].z += h;
        for(auto to : v[x]) {
            int y = to.first, w = b[to.second].l;
            if(end_time[to.second] == -1) {
                if(w < dis(a[x], a[y]))
                    end_time[to.second] = i;
            }
        }
    }

    // for(int i=1; i<=m; i++) cerr << end_time[i] << " \n"[i == m];

    for(int i=1; i<=m; i++) if(end_time[i] > 0) g[end_time[i]].push_back(i);

    int q; cin >> q;
    for(int i=1; i<=q; i++) cin >> que[i].first >> que[i].second;

    set<int> qu;
    for(int i=1; i<=q; i++) qu.insert(i);
  
    dtr.build(n);
    for(int i=1; i<=m; i++) if(end_time[i] == -1) dtr.merge(b[i].x, b[i].y);
    for(int i=k; i>=1; i--) {
        vector<int> del;
        for(int j : qu) if(ans[j] == 0) {
            if(dtr.find(que[j].first) == dtr.find(que[j].second)) {
                ans[j] = i;
                del.push_back(j);
            }
        }
        for(int x : del) qu.erase(x);
        for(int x : g[i]) dtr.merge(b[x].x, b[x].y);
    }

    for(int i=1; i<=q; i++) {
        if(ans[i] == k) cout << -1 << '\n';
        else cout << ans[i] + 1 << '\n';
    }
