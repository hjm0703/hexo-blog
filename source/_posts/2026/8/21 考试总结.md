## 题目概述

**T1** 直接就是一个数学题，因为你只需要知道两个线段垂直相当于他们的斜率的乘积为 $-1$ 。然后使用一个 `gp_hash_table` 就可以了。这道题 $ 30$ 分钟 AC （我认为的）， 但是最后由于使用了自己手写的 $\gcd$ 因此被卡常了。但是为什么普通 $n^3$ 啥优化都不加就可以过啊?

**T2** 最开始写的 DP 状态有问题，但是当时我觉得好像不能优化了？ （我是傻逼），然后就对着最后的 $ 20$ 分直接写容斥，关键是容斥小数据是正确的。大了之后就会偏离。最后 $ 85$ 分，是 $ 80$ 分暴力 DP 加上错误的容斥带来的 $ 5$ 分。最后这道题大概 $ 10 : 10$ 因为容斥一直不对就下一题了。

**T3** 一看到 LCA 都大概知道是大粪套。然后简单优化变成时间复杂度仅仅和 $a$ 和 $b$ 的距离相关，然后这里需要让每一个点加上自己 $siz$ 减去链上儿子的 $siz$ ，然后我口胡了一个使用树链剖分的做法，然后想起了郭总昨天说过的话:

> 大部分时候 $ 300$ 的组成并不是 $ 100 + 100 + 100$

于是果断放弃，（因为我的树链剖分做法还更屎），然后此时大概 $ 11 : 20$ 。我一直在思考的是如何让懒标记知道自己的在链上的儿子是什么，而居然没有想到把标记打到儿子上。

**T4** 直接写暴力，然后视图实现特殊性质 $A$, $B$ 。最终只拿了 $ 30$ 分。

template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }

static constexpr int N = 1e5 + 5, mod = 998244353, inv2 = (mod + 1) / 2;
int n, m, k;
int mul[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct Math {
    int ksm(int x, int y) const {
        int ans = 1;
        while(y) {
            if(y&1) ans = 1ll * ans * x % mod;
            x = 1ll * x * x % mod, y >>= 1;
        }
        return ans;
    }

    int mul[N], inv[N];
    void build(int n) {
        mul[0] = inv[0] = 1;
        for(int i=1; i<=n; i++) {
            mul[i] = 1ll * mul[i-1] * i % mod;
            inv[i] = ksm(mul[i], mod - 2);
        }
    }

    int binom(int n, int m) {
        if(n < m || n < 0 || m < 0) return 0;
        return 1ll * mul[n] * inv[m] % mod * inv[n-m] % mod;
    }
}mt;

int dp[505][505][505];
void solve() {
    dp[0][0][0] = 1;
    for(int i=0; i<n; i++) {
        for(int j=0; j<=k; j++) {
            for(int k=0; k<=n; k++) {
                if(!dp[i][j][k]) continue;
                if(j != ::k) {
                    // cerr << i + 1 << ' ' << j + 1 << ' ' << k << '\n';
                    dp[i+1][j+1][k] = (dp[i+1][j+1][k] + dp[i][j][k]) % mod;
                }
                if(k != n) {
                    dp[i+1][0][k+1] = (dp[i+1][0][k+1] + dp[i][j][k]) % mod;
                }
            }
        }
    }
}

int dp2[5005][2][5005];
void solve2() {

                    // cerr << i + 1 << ' ' << j + 1 << ' ' << k << '\n';
                    dp2[i+1][j+1][k] = (dp2[i+1][j+1][k] + dp2[i][j][k]) % mod;

                    dp2[i+1][0][k+1] = (dp2[i+1][0][k+1] + dp2[i][j][k]) % mod;
                }
            }
        }
    }
}

    else if(k == 1) solve2();
  
    // 实际上会乘上的权值
    int val = 1ll * m % mod * ((m + 1) % mod) % mod * inv2 % mod;

    mul[0] = 1;
    for(int i=1; i<=n; i++) mul[i] = 1ll * mul[i-1] * (m % mod) % mod;

            sum = 0;
            for(int j=0; j<=k; j++) sum = (sum + dp[n][j][i]) % mod;

                int add = 0;
                for(int t=1; t*j<=n; t++) {
                    int sign = (t & 1) ? 1 : -1;
                    int last = n - t * j;
                    add = (add + sign * (1ll * mt.binom(last + t, t) * mt.binom(last, i) % mod) + mod) % mod;
                }
                sum = (sum + sign * add + mod) % mod;
            }
        }

        // cerr << sum << '\n';
      
        // cerr << i << ' ' << w << ' ' << sum << ' ' << mul[n-i] << '\n';

        ans = (ans + 1ll * w * sum % mod * mul[n-i] % mod) %  mod;
    }

    cout << ans << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
#define int long long

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 2e5 + 5, mod = 1e9 + 7;
int n, m;
int du[N];
vector<int> v[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct Tree {
    int idx, dfn[N], siz[N];
    int dep[N], fa[N][20];

    void build(int x, int f) {
        dfn[x] = ++idx, siz[x] = 1;
        fa[x][0] = f, dep[x] = dep[f] + 1;
        for(int i=1; i<=19; i++) fa[x][i] = fa[fa[x][i-1]][i-1];
        for(int y : v[x]) if(y != f) {
            build(y, x), siz[x] += siz[y];
        }
    }

    int lca(int x, int y) {
        if(dep[x] < dep[y]) swap(x, y);
        for(int i=19; i>=0; i--)
            if(dep[fa[x][i]] >= dep[y])
                x = fa[x][i];
        if(x == y) return x;
        for(int i=19; i>=0; i--)
            if(fa[x][i] != fa[y][i])
                x = fa[x][i], y = fa[y][i];
        return fa[x][0];
    }
}tr;

namespace Sub1 {
    int ans[N], laz[N];
    // 这里是最基础的暴力

    void MakeTree(int x, int w) { // 处理一个子树中互相
        laz[x] += w;
    }

    void dfs(int x, int fa) {
        for(int y : v[x]) if(y != fa) {
            laz[y] += laz[x];
            dfs(y, x);
        }

        ans[x] += laz[x];
        for(int y : v[x]) if(y != fa) {
            ans[x] += 1ll * tr.siz[y] * laz[x] % mod * 2 % mod;
            ans[x] += 1ll * (tr.siz[x] - tr.siz[y] - 1) * tr.siz[y] % mod * laz[x] % mod;
            ans[x] %= mod;
        }
        ans[x] %= mod;
    }
  
    void solve() {
        for(int i=1; i<=m; i++) {
            int a, b, w; cin >> a >> b >> w;
            if(tr.dfn[b] <= tr.dfn[a] && tr.dfn[a] <= tr.dfn[b] + tr.siz[b] - 1) {
                swap(a, b);
            }
          
            if(tr.dfn[a] <= tr.dfn[b] && tr.dfn[b] <= tr.dfn[a] + tr.siz[a] - 1) {
                MakeTree(b, w);
                if(a == b) continue;
                int p = tr.fa[b][0], last = b;
                do {
                    ans[p] += 1ll * (tr.siz[p] - tr.siz[last]) * tr.siz[b] % mod * w % mod;
                    ans[p] %= mod;
                    last = p, p = tr.fa[p][0];
                } while(last != a);
            } else {
                int l = tr.lca(a, b);
                ans[l] += 1ll * tr.siz[a] * tr.siz[b] % mod * w % mod;
                ans[l] %= mod;
            }
        }

        dfs(1, 0);

        for(int i=1; i<=n; i++) cout << (ans[i] % mod + mod) % mod << " \n"[i == n];
    }
}

namespace Sub2 {
    int ans[N], laz[N], laz2[N];

    void MakeTree(int x, int w) { // 处理一个子树中互相
        laz[x] += w;
    }

    void dfs(int x, int fa) {
        for(int y : v[x]) if(y != fa) {
            laz[y] += laz[x], laz2[y] += laz2[x];
            dfs(y, x);
        }

        ans[x] += laz[x] + laz2[x];
        for(int y : v[x]) if(y != fa) {
            ans[x] += 1ll * tr.siz[y] * laz[x] % mod * 2 % mod;
            ans[x] += 1ll * (tr.siz[x] - tr.siz[y] - 1) * tr.siz[y] % mod * laz[x] % mod;
            ans[x] %= mod;
        }
        ans[x] %= mod;
    }
  
    void solve() {
        for(int i=1; i<=m; i++) {
            int a, b, w; cin >> a >> b >> w;
            if(tr.dfn[b] <= tr.dfn[a] && tr.dfn[a] <= tr.dfn[b] + tr.siz[b] - 1) {
                swap(a, b);
            }
          
            if(tr.dfn[a] <= tr.dfn[b] && tr.dfn[b] <= tr.dfn[a] + tr.siz[a] - 1) {
                MakeTree(b, w);
                if(a == b) continue;
                laz2[a] += 1ll * w * tr.siz[b] % mod, laz2[b] -= 1ll * w * tr.siz[b] % mod;
                laz2[a] %= mod, laz2[b] %= mod;
            } else {
                ans[tr.lca(a, b)] += 1ll * tr.siz[a] * tr.siz[b] % mod * w % mod;
            }
        }

        dfs(1, 0);

        for(int i=1; i<=n; i++) cout << (ans[i] % mod + mod) % mod << " \n"[i == n];
    }
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    #ifdef LOCAL
    freopen("samples/input.txt", "r", stdin);
    freopen("samples/merge5.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("merge.in", "r", stdin);
    freopen("merge.out", "w", stdout);
  
    cin >> n >> m;
    for(int i=1; i<n; i++) {
        int x, y; cin >> x >> y;
        v[x].push_back(y);
        v[y].push_back(x);
        du[x]++, du[y]++;
    }

    tr.build(1, 0);

    bool flag = true;
    for(int i=1; i<=n; i++) flag &= (du[i] <= 2);
  
    if(n <= 5000 && m <= 5000) Sub1::solve();
    else if(flag) Sub2::solve();
    else Sub1::solve();
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 205, mod = 998244353;
int c, n, T;
int a[N], t[N];

int top, ans[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    #ifdef LOCAL
    freopen("samples/input.txt", "r", stdin);
    freopen("samples/dream5.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("dream.in", "r", stdin);
    freopen("dream.out", "w", stdout);
  
    cin >> c >> n >> T;
    for(int i=1; i<=n; i++) cin >> a[i];
    for(int i=1; i<=n; i++) cin >> t[i];

    reverse(a+1, a+n+1);
    reverse(t+1, t+n+1);

    if((1 <= c && c <= 4) || (n <= 20)) {
        for(int i=1; i<=n; i++) {
            int w = 1 << i, ran = 0;
            for(int j=0; j<w; j++) {
                vector<pair<int, int>> v;
              
                for(int k=i; k>=1; k--) {
                    if(j & (1 << (k-1)))
                        v.emplace_back(a[k], t[k]);
                }
              
                for(int k=i; k>=1; k--) {
                    if(!(j & (1 << (k-1))))
                        v.emplace_back(a[k], t[k]);
                }

                int  time = 0, nans = 1;
                for(auto item : v) {
                    time += item.second;
                    if(time <= T) nans = 1ll * nans * (item.first + 1) % mod;
                    else break;
                }
              
                ran = (ran + nans - 1) % mod;
            }

            ans[++top] = ran;
        }

        for(int i=top; i>=1; i--) cout << ans[i] << " \n"[i == 1];
    } else if((5 <= c && c <= 7) || (10 <= c && c <= 13)) {
        for(int i=1; i<=n; i++) {
            int w = 1, res = 1;
            for(int j=1; j<=i; j++) w = 1ll * w * 2 % mod;
            for(int j=1; j<=i; j++) res = 1ll * res * (a[j] + 1) % mod;
            ans[++top] = 1ll * w * (res - 1 + mod) % mod;
        }

        for(int i=top; i>=1; i--) cout << ans[i] << " \n"[i == 1];
    }
    return 0;
}
