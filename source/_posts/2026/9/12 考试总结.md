## 题目概述

**省流：** $ 30 + 70 + 50 + 40 = 190$ ， 依旧没有 A 题。

首先场上开到这个 T1 一开就发现非常的诈骗。因为在树的情况下，规定了连通块的个数，就最多只有一种情况。然后 $ 25$ 分钟过了之后直接 **T2** 了。但是其中出现了两个问题：

- 计算子树大小的时候没有加上自己的 $ 1$。
- 被卡常了， 单纯时间复杂度是 $ 2 \times 10^8$ ，但是注意到： `请注意常数因子对程序效率带来的影响。`。实际上正解的时间复杂度确实更优一些。

然后第一眼贪心， 第二眼 DP。后来我写 DP 的时候实际上认为是可以通过的。因为我最开始误认为每一个 $x$ 不会超过 $ \sqrt{c}$ ，然后后来发现伪完了。然后加了个优化之后就有 $ 70$ 分了。我就先看 **T3** 了。

然后 **T3** 本身就是一个 **支配树** 板子题。但是我遇到了如下问题：

- 这可是 CSP - S 模拟赛， 怎么可能考支配树呢？
- 最开始我认为对于 DAG 上有多个根的情况，如果有多个跟可以到达当前处理的点， 就都需要计入贡献。为什么我一直绕不过去呢。 因为最开始我就把能够产生贡献的条件转化为了是当前节点的一个支配点。然后我一寻思，这里两个根是不是都算是支配点。然后我就炸了。

实际上我就一直没有尝试去写过支配树的代码。因为首先是写着写着就发现了我刚才的问题，根本写不下去。并且大样例也无法完全证明正确性（这里最大的大样例是一颗树）。

**T4** 是原题。然后基本结论还记得。然后优化的时候基本上能想到的就是 $\mathcal O(n \sqrt{n} \log n)$ 的做法。然后实际上正解是每一次加入所有能够加入的。然后可以发现如果以前的答案为 $ mex$， 那么新加入如果产生变化，必然增加一个在以前答案和现在之间的值，这个是 $ \mathcal O(n \log^2 n)$ 的。但是这种复杂度真的有点难想。

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

int siz[N];
void build(int x, int fa) {
    siz[x] = 1;
    for(int y : v[x]) if(y != fa) {
        build(y, x);
        siz[x] += siz[y];
    }
}

bool check(int x, int fa, int lim) {
    bool res = true;
    int sum = 0;
    for(int y : v[x]) if(y != fa) {
        res &= check(y, x, lim);
        sum += siz[y] % lim;
    }
    if(sum > lim) res = false;
    return res;
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

            ans += check(1, 0, i);
            if(i != n / i) ans += check(1, 0, n / i);
        }
    }

#include <bits/stdc++.h>

template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 1e5 + 5, M = 1e6 + 5, K = 1e6 + 5;
int n, c;
ll pr[K], sf[K];
ll mx, dp[2][K];
long long a[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
  
    #ifdef LOCAL
    freopen("samples/input2.txt", "r", stdin);
    freopen("samples/seq4.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("seq.out", "w", stdout);
  
    cin >> n >> c;
    for(int i=1; i<=n; i++) cin >> a[i], mx = max(mx, (ll) a[i]);
  
    int b = max((ll) c, mx) + 5;

    if(n >= 10) b = min(b, 2000ll);

    memset(dp, 0x3f, sizeof dp);
    int now = 1, pre = 0;
    for(int j=0; j<=b; j++) dp[now][j] = j * j;

    memset(pr, 0x3f, sizeof pr);
    memset(sf, 0x3f, sizeof sf);
    for(int j=0; j<=b; j++) {
        ll nw = dp[now][j] + c * (b - j);
        if(j == 0) pr[j] = nw;
        else pr[j] = min(pr[j - 1], nw);
    }
  
    for(int j=b; j>=0; j--) {
        ll nw = dp[now][j] + c * j;
        if(j == b) sf[j] = nw;
        else sf[j] = min(sf[j + 1], nw);
    }

    int last_b = b;
    for(int i=2; i<=n; i++) {
        now ^= 1, pre ^= 1;
        for(int j=0; j<=b; j++) {
            int p = a[i] + j - a[i - 1];

            if(p < 0) {
                dp[now][j] = sf[0] + - c * p + j * j;
            } else if(p > last_b) {
                dp[now][j] = pr[last_b] + c * (p - last_b) + j * j;
            } else {
                dp[now][j] = min(sf[p] - c * p, pr[p] - c * (last_b - p)) + j * j; 
            }
        }

        last_b = b;
        for(int j=0; j<=b; j++) {
            ll nw = dp[now][j] + c * (b - j);
            if(j == 0) pr[j] = nw;
            else pr[j] = min(pr[j - 1], nw);
        }
      
        for(int j=b; j>=0; j--) {
            ll nw = dp[now][j] + c * j;
            if(j == b) sf[j] = nw;
            else sf[j] = min(sf[j + 1], nw);
        }
    }

    ll ans = 0x3f3f3f3f3f3f3f3f;
    for(int i=0; i<=last_b; i++) ckmin(ans, dp[now][i]);
    cout << (long long) ans << '\n';

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

#include <bits/stdc++.h>
#include <assert.h>
using namespace std;

using ll = long long;

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 2e5 + 5;
int n, m;
bitset<1005> vis[1005];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct Tree {
    vector<int> v[N];
    int idx, dfn[N];
    int dep[N], fa[N][20];

    void build(int x, int f) {
        dfn[x] = ++idx;
        dep[x] = dep[f] + 1;
        fa[x][0] = f;
        for(int i=1; i<=19; i++) fa[x][i] = fa[fa[x][i - 1]][i - 1];
        for(int y : v[x]) if(y != f && !dep[y]) {
            build(y, x);
        }
    }

    int lca(int x, int y) {
        if(dep[x] < dep[y]) swap(x, y);
        for(int i=19; i>=0; i--)
            if(dep[fa[x][i]] >= dep[y])
                x = fa[x][i];
        if(x == y) return x;
        for(int i=19; i>=0; i--)
            if(fa[x][i] != fa[y][i])
                x = fa[x][i], y = fa[y][i];
        return fa[x][0];
    }
} tr;

struct BIT {
    int n, c[N];

    void build(int _n) { n = _n; memset(c, 0, sizeof c); }
    void add(int x, int k) { while(x && x <= n) c[x] += k, x += (x & -x); }
    int ask(int x) { int ans = 0; while(x) ans += c[x], x -= (x & -x); return ans; }
} btr;

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
  
    #ifdef LOCAL
    freopen("samples/input.txt", "r", stdin);
    freopen("samples/set4.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("set.in", "r", stdin);
    freopen("set.out", "w", stdout);
  
    cin >> n;

    if(n <= 1000) {
        for(int i=1; i<=n; i++) {
            int c; cin >> c;
            if(c) vis[i].set();
            while(c --> 0) {
                int x; cin >> x;
                vis[i] &= vis[x];
            }
            vis[i][i] = true;
        }

        cin >> m;
        for(int i=1; i<=m; i++) {
            int c; cin >> c;
            bitset<1005> ans;
            while(c --> 0) {
                int x; cin >> x;
                ans |= vis[x];
            }
            cout << ans.count() << '\n';
        }
    } else { // 所有 c 都为 0 / 1 的部分分
        for(int i=1; i<=n; i++) {
            int c; cin >> c;
            while(c --> 0) {
                int x; cin >> x;
                tr.v[x].push_back(i);
            }
        }

        for(int i=1; i<=n; i++) if(!tr.dep[i]) tr.build(i, 0);
      
        cin >> m;
        for(int i=1; i<=m; i++) {
            int c; cin >> c;
            vector<int> v;
            while(c --> 0) { int x; cin >> x; v.push_back(x); }
            sort(v.begin(), v.end(), [&] (int x, int y) { return tr.dfn[x] < tr.dfn[y]; });

            int ans = 0, last = -1;
            for(auto x : v) {
                ans += tr.dep[x];
                if(last != -1) {
                    int lc = tr.lca(last, x);
                    if(lc) ans -= tr.dep[lc];
                }
                last = x;
            }

            cout << ans << '\n';
        }
    }

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

using ll = long long;

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 1e5 + 5;
int n, m;
int a[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

struct SegmentTree {
    #define lc (p << 1)
    #define rc (p << 1 | 1)

    pair<int, int> mx[N<<2];

    void change(int p, int l, int r, int x, pair<int, int> k) {
        if(l == r) return mx[p] = k, void();
        int mid = (l + r) >> 1;
        if(x <= mid) change(lc, l, mid, x, k);
        else change(rc, mid + 1, r, x, k);
        mx[p] = min(mx[lc], mx[rc]);
    }

    pair<int, int> query(int p, int l, int r, int x, int y) {
        if(x <= l && r <= y) return mx[p];
        int mid = (l + r) >> 1;
        pair<int, int> ans = {0x3f3f3f3f, -1};
        if(x <= mid) ckmin(ans, query(lc, l, mid, x, y));
        if(y > mid) ckmin(ans, query(rc, mid + 1, r, x, y));
        return ans;
    }
} T;

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    #ifdef LOCAL
    freopen("samples/input2.txt", "r", stdin);
    freopen("samples/mex3.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("mex.in", "r", stdin);
    freopen("mex.out", "w", stdout);
  
    cin >> n;
    for(int i=1; i<=n; i++) cin >> a[i], T.change(1, 1, n, i, {a[i], i});
    cin >> m;
    while(m --> 0) {
        int l, r; cin >> l >> r;
        vector<int> v;
      
        int mex = 1;
        for(int i=l; i<=r; i++) {
            int x = T.query(1, 1, n, l, r).second;
            if(a[x] <= mex) mex += a[x];
            else break;
            T.change(1, 1, n, x, {0x3f3f3f3f, i});
            v.push_back(x);
        }

        cout << mex << '\n';

        for(auto i : v) T.change(1, 1, n, i, {a[i], i});
    }

    cerr << 1.0 * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}
