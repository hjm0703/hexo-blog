# 题目概述

省流： $ 90 + 50 + 75 + 65 = 280$ ，无题 AC 😢。

开始看 **T1** ，看上去就非常困难，从前往后做？ 各种颜色的去重问题对吧。然后我就相当可以只处理这些修改的线段的交点， 随机有想到如果我们把最后修改的线段去除掉（去掉一行/列）以前连接的就可以连起来。然后递归求解。终于发现绕了一些些弯路，可以直接从后往前加。 最后 $ 90$ 是因为有时候一列都被填完了，就会在答案上加 $ 0$。 给答案总数造成影响。

**8:50** 开 **T2** ，发现这个东西怎么那么像广告中的那种小游戏呢，但是他有 $ m$ 的限制，贪心就很难了。然后注意到题目 $\mathcal O(nV)$ 是可以通过的，因此考虑 DP。然后就发现每一个位置的转移实际上实际上转移到的是一个区间。因此就使用单调队列优化就可以了。**9:20** 过大样例，然后写了一个对拍。对拍的时候直接吓哭了，因为我把单调队列写成了单调栈。最后唯一对拍的题目是挂的最惨的。 我的单调队列 `l = 1, r = 0` 没有写，但是根据区间的影响范围，每一个有刚好会有 `l = r + 1`，因此小数据没有问题，交上去 $ 50$ 分 RE 了。

**10:00** 开 T3， 这个基础的容斥还是很好写的，当然最开始容斥想复杂了。然后看到整除，直接上整除分块就 TLE 了， 实际上可以枚举得到的因子，然后反推出其影响的范围。本质时间复杂度差不多 $\mathcal O(n \sqrt{n} + n \log n)$ 和 $\mathcal O(n \log^2 n)$ ，但是整除分块需要卡常。

**10:40** 开 T4， 写不动了。稍微枚举了一下发现这里 $k$ 虽然很大，但是大部分都是循环的。具体来说我直接按照两个一循环。然后其他的暴力，但是很明显暴力是过不了的，因此我有加上了经典卡时技巧。每一次之更新上一次更新周围的。然后就有 $ 65$ ~ $ 70$ 分了。这里之所以~~只有~~真么一点实际上是因为对于只有两个字母的部分是不存在循环节的，而实际上是整个区间在平移。因此这个做法在这个部分分中完全是错误的。

前三道题都是非常好写的，但是第 $ 4$ 题一遇到循环节的题目就非常的不好做。

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
#define int long long

template<class T> inline void ckmin(T& x, T y) { if(x > y) x = y; }
template<class T> inline void ckmax(T& x, T y) { if(x < y) x = y; }
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 2e5 + 5, mod = 998244353;
int n;
int a[N], ans[N];
int c[N], d[N];
int e[N], f[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

int ksm(int x, int y) {
    int ans = 1;
    while(y) {
        if(y&1) ans = 1ll * ans * x % mod;
        x = 1ll * x * x % mod, y >>= 1;
    }
    return ans;
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    #ifdef LOCAL
    freopen("samples/input3.txt", "r", stdin);

    freopen("samples/output.txt", "w", stdout);

    freopen("gcdseq.out", "w", stdout);
  
    cin >> n;
    for(int i=1; i<=n; i++) cin >> a[i];

    int mi = 1e9;
    for(int i=1; i<=n; i++) mi = min(mi, a[i]);

    if(n * mi <= 100000) { // 先打一个暴力
        int res = 0;
        for(int i=mi; i>=1; i--) {
            ans[i] = 1; int sum = 1;
            for(int j=1; j<=n; j++) sum = 1ll * sum * (a[j] / i - 1) % mod;
            for(int j=1; j<=n; j++) ans[i] = 1ll * ans[i] * (a[j] / i) % mod;
            ans[i] = (ans[i] - sum + mod) % mod;
            res = (res + ans[i]) % mod;
        }
        cout << res << '\n';
    } else {
        c[0] = d[0] = e[0] = f[0] = 1;
        for(int i=1; i<=mi; i++) c[i] = d[i] = e[i] = f[i] = 1;
      
        for(int i=1; i<=n; i++) {
            for(int l=1, r; l<=mi; l=r+1) {
                r = a[i] / (a[i] / l);
                c[l] = 1ll * c[l] * (a[i] / l) % mod;
                e[r + 1] = 1ll * e[r + 1] * (a[i] / l) % mod;
                d[l] = 1ll * d[l] * (a[i] / l - 1) % mod;
                f[r + 1] = 1ll * f[r + 1] * (a[i] / l - 1) % mod;
            }
        }

        int res = 0;
        for(int i=1; i<=mi; i++) {
            c[i] = 1ll * c[i-1] * c[i] % mod;
            d[i] = 1ll * d[i-1] * d[i] % mod;
            e[i] = 1ll * e[i-1] * e[i] % mod;
            f[i] = 1ll * f[i-1] * f[i] % mod;
            ans[i] = (1ll * c[i] * ksm(e[i], mod - 2) % mod - 1ll * d[i] * ksm(f[i], mod - 2) % mod + mod) % mod;
            res = (res + ans[i]) % mod;
        }
        cout << res << '\n';
    }

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

struct FastIO {

    template<class T> inline FastIO& operator >> (T& x) {
        char ch = getc(); int f = (x = 0);

        return x = f ? -x : x, *this;
