## 题目概述

**省流：** 考场分数: $100 + 100 + 100 + 24$ 。

首先看到 **T1** 是很顺的，因为看到最大值直接枚举最大值，然后维护最大的 $K$ 个最大就可以了。

然后写 **T2**， 实际上最开始各种通过转化转化成了需要通过各种代价的区间覆盖，然后这个就直接暴力就可以了。但是实际上对于这种经典的区间覆盖问题，我们还可以用下面的办法：

> 对于所有区间上的点作为图上的点，然后一个区间相当于从最左边向最右边连边。
> 
> 然后由于区间不应顶需要端点正好重合在一起，而是可以部分重叠。因此我们对于每一个点连向他前面一个代价为 $0$ 的边。
> 
> 最后为了处理中间有一段不需要连边的情况，我们在哪里对于每一个节点连向他们下一个节点就可以了。
> 
> 最后直接在图上跑一个最短路，可以发现时间复杂度是正确的。

看到 **T3** 的时候，基础的 DP 还是很好想的，但是由于第二题都不是一个 **最** 普通的 DP。并且这个东西看上去复杂度有一些奇怪，因此总是往其他地方去想。

因此我直接发现这里的所有转移都可以变化为 **卷积** + **加法** + **区间平移**，因此直接使用一个 **NTT** 去维护。但是 NTT 怎么写啊？ 好久没写就忘完了 😢。

写 **T4** 的时候因为看到取模，就一直往数学的方向想，最开始吧基础的暴力写完之后发现最大的数据大概 5s 多，因此直接思考如果卡常。最开始我发现如果当前维护的两个变量形式是 $x, y$ ，那么最终得到的答案的式子也一定是 $ax + by$ 的形式，那么我们是不是只需要判断这个数是否能够在 $\bmod p$ 的情况下为 $r$ 和 $s$ 的形式就可以了。即裴蜀定理，当时我真的非常的确幸这样子就能卡过，直到我发现 $\gcd(x, y)$ 始终为 $1$ 。

后来后尝试在是否真的存在解 $a \ge 0, b \ge 0$ 的情况，然后通过那个通解公式去判断，然后实际上他们也是都有解的。~~应该吧~~

题目正解是折半搜索，因为正解发现 $x / y$ 只要选择不一样，最终的结果区间也一定不会有重复的。但是实际上有更简单的可以过的做法，就是把最简单的暴力中所有的询问一起 DFS。

实际上这个题目最开始就有一种直觉，因为 $x, y$ 增长趋势很快，因此大致也是指数级的，阴恻最多总共点的数量也不会很多，然后说不定直接搜索就可以了。

#include <bits/stdc++.h>
using namespace std;

using ll = long long;
#define int long long
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr ll inf = 0x3f3f3f3f3f3f3f3fll;
static constexpr int N = 1e5 + 5;
int n;
int a[N], c[N], r[N];
vector<pair<int, int>> v[N]; // 以 i 为右端点的区间
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

ll dp[N];
int last[N]; // 左边最后一个区间右端点

int type[N]; // 0: 无需覆盖, 1: 区间左端点, 2: 区间中间, 3: 区间右端点
int sum1[N], sum2[N];  // 每一个点是否被覆盖
int tot, b[N], cnt[N]; // 离散化所需要的数组

struct SegmentTree {
    #define lc (p << 1)
    #define rc (p << 1 | 1)

    ll mi[N<<2];

    void change(int p, int l, int r, int x, ll k) {
        if(l == r) return mi[p] = k, void();
        int mid = (l + r) >> 1;
        if(x <= mid) change(lc, l, mid, x, k);
        else change(rc, mid+1, r, x, k);
        mi[p] = min(mi[lc], mi[rc]);
    }

    ll query(int p, int l, int r, int x, int y) {
        if(x > y) return inf;
        if(x<=l && r<=y) return mi[p];
        int mid = (l + r) >> 1; ll ans = inf;
        if(x<=mid) ans = min(ans, query(lc, l, mid, x, y));
        if(y>mid) ans = min(ans, query(rc, mid+1, r, x, y));
        return ans;
    }
}T;

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

    #ifdef LOCAL

    // freopen("samples/anchor1.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("anchor.in", "r", stdin);
    freopen("anchor.out", "w", stdout);
  
    cin >> n;
    for(int i=1; i<=n; i++) cin >> a[i];
    for(int i=1; i<=n; i++) cin >> c[i];
    for(int i=1; i<=n; i++) cin >> r[i];

    // ll sum = 0;
    // for(int i=1; i<=n; i++) sum += r[i];
    // cerr << sum << '\n';
  
    for(int i=1; i<=n; i++) b[++tot] = a[i];
    sort(b+1, b+tot+1);
    for(int i=1; i<=n; i++) {
        int x = lower_bound(b+1, b+tot+1, a[i]) - b;
        a[i] = x + cnt[x], cnt[x]++;
    }

    for(int i=1; i<=n; i++) {
        if(a[i] < i) sum1[a[i]]++, sum2[i+1]++;
        if(a[i] > i) sum1[i]++, sum2[a[i]+1]++;
    }

    for(int i=1, sum=0; i<=n+1; i++) {
        sum -= sum2[i];
        if(sum == 0 && sum2[i]) type[i-1] = 3;
        sum += sum1[i];
        if(sum != 0 && sum == sum1[i]) type[i] = 1;
        if(sum && !type[i]) type[i] = 2;
    }

    // for(int i=1; i<=n; i++) cerr << type[i] << " \n"[i == n];

    for(int i=1; i<=n; i++) {
        if(type[i] == 3) last[i] = i;
        else last[i] = last[i-1];
    }
  
    for(int i=1; i<=n; i++) {
        int l = max(i - ::r[i] + 1, 0ll), r = min(i + ::r[i] - 1, n);
        if(l == r) continue;
        v[r].emplace_back(l, c[i]);
    }

    for(int i=1; i<=n; i++) {
        dp[i] = inf;
        for(auto range : v[i]) {
            int l = range.first, w = range.second, al = 0; // 这里 al 为查询最左边的点
            if(type[l] == 2) al = l;
            else al = last[l];
            // cerr << i << ' ' << al << ' ' << w << '\n';
            dp[i] = min(dp[i], T.query(1, 0, n, al, i-1) + w);
            // for(int j=al; j<i; j++) dp[i] = min(dp[i], dp[j] + w);
        }
        T.change(1, 0, n, i, dp[i]);
    }

    ll ans = inf;
    for(int i=n; i>=0; i--) {
        ans = min(ans, dp[i]);
        if(type[i] == 3) break;
    }

    if(ans < inf) cout << ans << '\n';
    else cout << -1 << '\n';

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
static constexpr int N = 5005, mod = 998244353;
int n, k;
vector<int> v[N];
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

int siz[N], son[N];
void build(int x, int fa) {
    siz[x] = 1;
    for(int y : v[x]) if(y != fa) {
        build(y, x), siz[x] += siz[y];
        if(son[x] == 0 || siz[son[x]] < siz[y]) son[x] = y;
    }
}

int dp[N][2][N], nxt[2][N];
void solve(int x, int fa) {
    if(v[x].size() - (fa > 0) == 0) {
        dp[x][0][0] = 1;
        dp[x][1][0] = 0;
        return ;
    }

    dp[x][0][0] = 1;
    dp[x][1][0] = 0;
  
    int mx = 1;
    for(int y : v[x]) if(y != fa) {
        solve(y, x);
        memset(nxt, 0, sizeof nxt);
        for(int t : {0, 1}) for(int g : {0, 1}) {
            for(int k : {1, 2}) {
                for(int i=mx; i>=0; i--) {
                    if(dp[x][t][i] == 0) continue;
                    for(int j=siz[y]; j>=0; j--) {
                        if(dp[y][g][j] == 0) continue;
                        int nt = t | (k == 1), ng = g | (k == 2);
                        int ni = i + j + (nt != t) + (ng != g);
                        nxt[nt][ni] = (nxt[nt][ni] + 1ll * dp[x][t][i] * dp[y][g][j] % mod) % mod;
                    }
                }
            }
        }
        mx += siz[y];
        memcpy(dp[x], nxt, sizeof dp[x]);
    }
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
  
    #ifdef LOCAL
    freopen("samples/input2.txt", "r", stdin);
    // freopen("samples/awake3.in", "r", stdin);
    freopen("samples/output.txt", "w", stdout);
    #endif

    freopen("awake.in", "r", stdin);
    freopen("awake.out", "w", stdout);
  
    cin >> n >> k;
    for(int i=1; i<n; i++) {
        int x, y; cin >> x >> y;
        v[x].push_back(y), v[y].push_back(x);
    }

    build(1, 0);
  
    for(int i=1; i<=n; i++) {
        sort(v[i].begin(), v[i].end(), [&] (int x, int y) {
            return siz[x] > siz[y];
        });
    }
  
    solve(1, 0);

    cout << (dp[1][0][k] + dp[1][1][k]) % mod << '\n';

    cerr << 1. * clock() / CLOCKS_PER_SEC << '\n';
    return 0;
}

#define ll long long
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
ll n, m, P, Q;
ll r, s;
/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/

inline ll query(ll dp0, ll dp1) {
    ll res = 0;

    if(dp0 % P == r && (dp0 + dp1) % P == s) {
        res++;
    }
    // for(int t : {0, 1}) {
    //     ll ndp0 = dp0 + dp1;
    //     ll ndp1 = 0;
    //     if(t == 0) ndp1 = dp0;
    //     else ndp1 = dp0 + ((dp1 + dp0) << 1);
    //     if(ndp0 > n || ndp0 + ndp1 > m) break;
    
    //     res += query(ndp0, ndp1);
    // }

    ll ndp0 = dp0 + dp1;
    ll ndp1 = dp0;
    if(ndp0 > n || ndp0 + ndp1 > m) return res;
  
    res += query(ndp0, ndp1);

    ndp0 = dp0 + dp1;
    ndp1 = dp0 + ((dp1 + dp0) << 1);
    if(ndp0 > n || ndp0 + ndp1 > m) return res;
  
    res += query(ndp0, ndp1);
  
    return res;
}

/*~~~~~~~~~~~~~~~~~~~~ Boundary Line ~~~~~~~~~~~~~~~~~~~~*/
